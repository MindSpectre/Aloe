cmake_minimum_required(VERSION 4.3)
project(r CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_SCAN_FOR_MODULES ON)
add_library(a STATIC)
target_sources(a PUBLIC FILE_SET modules TYPE CXX_MODULES BASE_DIRS ${CMAKE_CURRENT_SOURCE_DIR} FILES a.cppm)
add_library(opts INTERFACE)
target_include_directories(opts SYSTEM INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/inc)
target_compile_options(opts INTERFACE "SHELL:-include cfg.h -march=corei7 -DX=1")
add_library(b STATIC b.cpp)
target_link_libraries(b PUBLIC opts)
add_executable(main main.cpp)
target_link_libraries(main PRIVATE a b)
