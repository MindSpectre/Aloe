cmake_minimum_required(VERSION 4.3)
project(r CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_SCAN_FOR_MODULES ON)
add_library(a STATIC)
target_sources(a PUBLIC FILE_SET modules TYPE CXX_MODULES BASE_DIRS ${CMAKE_CURRENT_SOURCE_DIR} FILES a.cppm p.cppm)
add_library(l STATIC)
target_sources(l PUBLIC FILE_SET modules TYPE CXX_MODULES BASE_DIRS ${CMAKE_CURRENT_SOURCE_DIR} FILES l.cppm)
target_link_libraries(l PUBLIC a)
add_library(c INTERFACE)
target_link_libraries(c INTERFACE l)
add_executable(main main.cpp)
target_compile_options(main PRIVATE -march=corei7)
target_link_libraries(main PRIVATE a c)
